Exemplos de Consultas
Consulta Básica
SELECT * FROM clientes LIMIT 1000
Consulta com Filtro
SELECT * FROM pedidos 
WHERE data_pedido > '2023-01-01'
LIMIT 500
Dados Agregados
SELECT 
  categoria_produto,
  COUNT(*) AS total_produtos,
  AVG(preco) AS preco_medio
FROM produtos
GROUP BY categoria_produto
Junção de Tabelas
SELECT 
  p.id_pedido, 
  c.nome_cliente,
  p.data_pedido,
  p.valor_total
FROM pedidos p
JOIN clientes c ON p.id_cliente = c.id
LIMIT 1000
